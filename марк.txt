import { Repository } from 'typeorm';

@Injectable()
export class SeedService {
    constructor(
        @InjectRepository(OrderStatus)
        private readonly orderStatusRepository: Repository<OrderStatus>,
        @InjectRepository(PizzaType)
        private readonly pizzaTypeRepository: Repository<PizzaType>,
        @InjectRepository(UserRoleType)
        private readonly userRoleTypeRepository: Repository<UserRoleType>,
    ) {
        this.runSeeds();
    }

    async runSeeds() {
        await this.productCategory();
        await this.orderStatus();
        await this.userRoleType();
    }

    async userRoleType() {
        await this.userRoleTypeRepository.save([
            this.userRoleTypeRepository.create({
                id: 1,
                name: 'user',
            }),
            this.userRoleTypeRepository.create({
                id: 2,
                name: 'admin',
            }),
        ]);
    }

    async productCategory() {
        await this.pizzaTypeRepository.save([
            this.pizzaTypeRepository.create({
                id: 1,
                name: 'Веганская',
                color: '19ff19',
            }),
            this.pizzaTypeRepository.create({
                id: 2,
                name: 'Острая',
                color: '900020',
            }),
            this.pizzaTypeRepository.create({
                id: 3,
                name: 'Очень острая',
                color: '660000',
            }),
            this.pizzaTypeRepository.create({
                id: 4,
                name: 'Сезонная',
                color: '1047c7',
            }),
        ]);
    }

    async orderStatus() {
        const statuses = [
            { id: 1, name: 'Ожидает подтверждения' },
            { id: 2, name: 'Подтвержден' },
            { id: 3, name: 'Готовится' },
            { id: 4, name: 'Готов к выдаче' },
            { id: 5, name: 'Выдан' },
            { id: 6, name: 'Отменен' },
        ];

        for (const status of statuses) {
            const existing = await this.orderStatusRepository.findOneBy({
                id: status.id,
            });
            if (!existing) {
                await this.orderStatusRepository.save(status);
            }
        }
    }
}

async function bootstrap() {
    const app = await NestFactory.create(AppModule);

    app.enableCors({ origin: true });

    app.useGlobalPipes(
        new ValidationPipe({
            whitelist: true, // Удаляет все неописанные поля
            forbidNonWhitelisted: false, // Выбрасывает ошибку для неизвестных полей
            transform: true,
            transformOptions: {
                exposeUnsetFields: false,
                enableImplicitConversion: true,
            },
        }),
    );

    // Swagger виден только если это не прод
    const config = new DocumentBuilder()
        .setTitle('Diplom')
        .setDescription('')
        .setVersion('1.0')
        .addApiKey(
            {
                type: 'apiKey',
                name: 'Authorization',
                in: 'header',
                description: 'Enter your token',
            },
            'X-AUTH-TOKEN',
        )
        .build();
    const documentFactory = () => SwaggerModule.createDocument(app, config);
    SwaggerModule.setup('api', app, documentFactory);
    await app.listen(process.env.PORT ?? 3000);
}

import { Reflector } from '@nestjs/core';
import { UserRoleService } from 'src/userRole/userRole.service';

@Injectable()
export class RolesGuard implements CanActivate {
    constructor(
        private readonly reflector: Reflector,
        private readonly userRoleService: UserRoleService,
    ) {}

    async canActivate(context: ExecutionContext): Promise<boolean> {
        const roles = this.reflector.get<string[]>(
            'roles',
            context.getHandler(),
        );

        if (!roles) {
            return false;
        }

        const request = context.switchToHttp().getRequest();
        const user = request.user;

        const userRole = await this.userRoleService.getUserRole(user.id);

        return roles.some((role: string) => {
            return role === userRole;
        });
    }
}

import {
  isUUID,
  registerDecorator,
  ValidationArguments,
  ValidationOptions,
} from 'class-validator';

export function IsUUIDArray(validationOptions?: ValidationOptions) {
  return function (object: Object, propertyName: string) {
    registerDecorator({
      name: 'isUUIDArray',
      target: object.constructor,
      propertyName: propertyName,
      options: validationOptions,
      validator: {
        validate(value: any, args: ValidationArguments) {
          return (
            Array.isArray(value) &&
            value.every((item) => typeof item === 'string' && isUUID(item))
          );
        },
        defaultMessage(args: ValidationArguments) {
          return `${args.property} must be an array of UUIDs`;
        },
      },
    });
  };

import { Injectable } from '@nestjs/common';
import { InjectRepository } from '@nestjs/typeorm';
import { Repository } from 'typeorm';
import { PizzaAdditiveRelation } from './entities/pizza-additive-relations.entity';
import { PizzaAdditive } from './entities/pizza-additive.entity';
import { PizzaSizeRelation } from './entities/pizza-size-relations.entity';
import { PizzaSize } from './entities/pizza-size.entity';
import { PizzaType } from './entities/pizza-type.entity';
import { Pizza } from './entities/pizza.entity';

@Injectable()
export class PizzaService {
    constructor(
        @InjectRepository(Pizza)
        private readonly pizzaRepository: Repository<Pizza>,
        @InjectRepository(PizzaAdditiveRelation)
        private readonly pizzaAdditiveRelationRepository: Repository<PizzaAdditiveRelation>,
        @InjectRepository(PizzaAdditive)
        private readonly pizzaAdditiveRepository: Repository<PizzaAdditive>,
        @InjectRepository(PizzaSizeRelation)
        private readonly pizzaSizeRelationRepository: Repository<PizzaSizeRelation>,
        @InjectRepository(PizzaSize)
        private readonly pizzaSizeRepository: Repository<PizzaSize>,
        @InjectRepository(PizzaType)
        private readonly pizzaTypeRepository: Repository<PizzaType>,
    ) {}

    async getPizza() {
        return this.pizzaRepository.find({
            relations: {
                pizzaType: true,
                pizzaSizeRelation: {
                    pizzaSize: true,
                    pizzaAdditiveRelation: {
                        pizzaAdditive: true,
                    },
                },
            },
        });
    }

    async getPizzaById(id: string) {
        return this.pizzaRepository.findOne({
            where: { id },
            relations: {
                pizzaType: true,
                pizzaSizeRelation: {
                    pizzaSize: true,
                    pizzaAdditiveRelation: {
                        pizzaAdditive: true,
                    },
                },
            },
        });
    }
}
